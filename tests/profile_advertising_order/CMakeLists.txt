cmake_minimum_required(VERSION 3.20)
project(bobtail_advertising_order_test C)

# Configure-only regression test: ZMK initializes BLE and its split transport
# at the same priority, so app source order must survive the source replacement.
set(APPLICATION_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/upstream")
file(MAKE_DIRECTORY "${APPLICATION_SOURCE_DIR}/src")
set(ble_fixture "void fixture(void) {\n    LOG_DBG(\"advertising from %d to %d\", advertising_status, desired_adv);\n}\n")
if(BOBTAIL_TEST_BAD_HOOK STREQUAL "missing")
  set(ble_fixture "void fixture(void) {}\n")
elseif(BOBTAIL_TEST_BAD_HOOK STREQUAL "duplicate")
  string(APPEND ble_fixture "${ble_fixture}")
endif()
file(WRITE "${APPLICATION_SOURCE_DIR}/src/ble.c" "${ble_fixture}")
foreach(name before split after)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.c" "void ${name}(void) {}\n")
endforeach()

set(ble_path "${APPLICATION_SOURCE_DIR}/src/ble.c")
if(BOBTAIL_TEST_RELATIVE)
  set(ble_path "src/ble.c")
  # Relative paths are relative to this fixture's source directory. Mark the
  # synthetic relative path generated so configure need not create it there.
  set_source_files_properties("${ble_path}" PROPERTIES GENERATED TRUE)
endif()
add_library(app STATIC "${CMAKE_CURRENT_BINARY_DIR}/before.c")
target_sources(app PRIVATE "${ble_path}" "${CMAKE_CURRENT_BINARY_DIR}/split.c")
target_sources(app PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/after.c")
if(BOBTAIL_TEST_DUPLICATE_SOURCE)
  target_sources(app PRIVATE "${ble_path}")
endif()

if(NOT BOBTAIL_POLICY_CMAKE)
  set(BOBTAIL_POLICY_CMAKE "${CMAKE_CURRENT_LIST_DIR}/../../cmake/profile_advertising.cmake")
endif()
include("${BOBTAIL_POLICY_CMAKE}")
bobtail_apply_profile_advertising_policy()

get_target_property(actual app SOURCES)
set(expected "${CMAKE_CURRENT_BINARY_DIR}/before.c;${PROJECT_BINARY_DIR}/bobtail_ble.c;${CMAKE_CURRENT_BINARY_DIR}/split.c;${CMAKE_CURRENT_BINARY_DIR}/after.c")
if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "BLE source moved past split initialization: ${actual}")
endif()
file(READ "${APPLICATION_SOURCE_DIR}/src/ble.c" original)
if(NOT original STREQUAL ble_fixture)
  message(FATAL_ERROR "Dependency source was modified")
endif()
file(READ "${PROJECT_BINARY_DIR}/bobtail_ble.c" patched)
string(FIND "${patched}" "bobtail_profile_advertising_allowed()" gate)
string(FIND "${patched}" "bobtail_profile_refresh_advertising(void)" refresh)
if(gate LESS 0 OR refresh LESS 0)
  message(FATAL_ERROR "Advertising gate or refresh hook missing")
endif()
message(STATUS "PASS: BLE remains before split initialization; other sources and dependency unchanged")
